-----JavaScript-----

document.addEventListener("DOMContentLoaded", () => {
  // Get all cards
  const cards = document.querySelectorAll(".card");

  // Add mousemove effect for each card
  cards.forEach((card) => {
    card.addEventListener("mousemove", (e) => {
      // Get position of mouse relative to card
      const rect = card.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;

      // Calculate rotation values based on mouse position
      const multiplier = 10; // Adjust for rotation intensity
      const rotateY = (x / rect.width - 0.5) * multiplier;
      const rotateX = (y / rect.height - 0.5) * -multiplier;

      // Apply the rotation and 3D effect
      card.style.transform = `perspective(1000px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) translateY(-15px)`;

      // Add subtle shadow based on rotation
      updateCardShadow(card, rotateX, rotateY);

      // Add highlight effect based on mouse position
      updateHighlight(card, x, y);
    });

    // Reset card on mouse leave
    card.addEventListener("mouseleave", () => {
      card.style.transform = "translateY(-15px)";
      card.style.boxShadow = "0 25px 50px rgba(0, 0, 0, 0.3)";

      // Remove highlight
      const highlight = card.querySelector(".card-highlight");
      if (highlight) {
        highlight.remove();
      }
    });

    // Add click effect
    card.addEventListener("click", () => {
      card.classList.add("card-click");
      setTimeout(() => {
        card.classList.remove("card-click");
      }, 300);
    });
  });

  // Function to update card shadow based on rotation
  function updateCardShadow(card, rotateX, rotateY) {
    const shadowX = rotateY * 2;
    const shadowY = rotateX * 2;
    card.style.boxShadow = `${shadowX}px ${shadowY}px 30px rgba(0, 0, 0, 0.4), 
                               0 25px 50px rgba(0, 0, 0, 0.3)`;
  }

  // Function to add highlight effect based on mouse position
  function updateHighlight(card, x, y) {
    // Remove existing highlight
    const existingHighlight = card.querySelector(".card-highlight");
    if (existingHighlight) {
      existingHighlight.remove();
    }

    // Create highlight element
    const highlight = document.createElement("div");
    highlight.classList.add("card-highlight");

    // Set highlight styles
    highlight.style.position = "absolute";
    highlight.style.width = "150px";
    highlight.style.height = "150px";
    highlight.style.borderRadius = "50%";
    highlight.style.background =
      "radial-gradient(circle, rgba(255,255,255,0.15) 0%, rgba(255,255,255,0) 70%)";
    highlight.style.pointerEvents = "none";
    highlight.style.transform = "translate(-50%, -50%)";
    highlight.style.top = `${y}px`;
    highlight.style.left = `${x}px`;
    highlight.style.opacity = "0.8";

    // Add highlight to card
    card.appendChild(highlight);
  }

  // Animate the background shapes
  animateBackgroundShapes();
});

// Function to animate background shapes
function animateBackgroundShapes() {
  const shapes = document.querySelectorAll(".shape");

  // Add additional random movements to shapes
  shapes.forEach((shape, index) => {
    // Get current position
    const computedStyle = window.getComputedStyle(shape);
    let currentTop = parseInt(computedStyle.top);
    let currentLeft = parseInt(computedStyle.left);

    // If values are NaN (for centered shape), set defaults
    if (isNaN(currentTop)) currentTop = window.innerHeight / 2;
    if (isNaN(currentLeft)) currentLeft = window.innerWidth / 2;

    // Set random movement
    setInterval(() => {
      // Generate random movement values
      const moveX = (Math.random() - 0.5) * 20;
      const moveY = (Math.random() - 0.5) * 20;

      // Apply movement with transition
      shape.style.transition = "transform 8s ease-in-out";
      shape.style.transform = `translate(${moveX}px, ${moveY}px) scale(${
        1 + Math.random() * 0.2
      })`;
    }, 8000);
  });
}

// Card button effects
document.addEventListener("DOMContentLoaded", () => {
  const buttons = document.querySelectorAll(".card-button");

  buttons.forEach((button) => {
    button.addEventListener("mouseover", () => {
      button.style.transform = "translateY(-5px)";
    });

    button.addEventListener("mouseout", () => {
      button.style.transform = "translateY(0)";
    });

    button.addEventListener("click", (e) => {
      // Create ripple effect
      const ripple = document.createElement("span");
      ripple.classList.add("ripple");
      button.appendChild(ripple);

      const rect = button.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;

      ripple.style.left = `${x}px`;
      ripple.style.top = `${y}px`;

      setTimeout(() => {
        ripple.remove();
      }, 600);

      // Alert for demo purposes
      alert("Button clicked: " + button.textContent);
    });
  });
});
